rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Впиши ваши две почты (те же, что в config.js), маленькими буквами
    function isUs() {
      return request.auth != null
        && request.auth.token.email in ['roza54468542@gmail.com', 'paskonastya05@gmail.com'];
    }

    match /{document=**} {
      allow read, write: if isUs();
    }
    // бот «Лимитов»: в 00:00 записывает профит P2P за день — только свои записи p2p-ДАТА
    match /tx/{id} {
      allow read, write: if request.auth != null
        && request.auth.token.email == 'p2p-bot@traty.app'
        && id.matches('p2p-[0-9]{4}-[0-9]{2}-[0-9]{2}');
    }
  }
}
